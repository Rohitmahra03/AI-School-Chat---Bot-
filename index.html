import express from "express"; import Groq from "groq-sdk"; import dotenv from "dotenv"; import path from "path"; import { fileURLToPath } from "url"; dotenv.config(); const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json({ limit: "2mb" })); app.use(express.static(path.join(__dirname, "public"))); if (!process.env.GROQ_API_KEY) { console.error("❌ GROQ_API_KEY missing in .env file"); } const groq = new Groq({ apiKey: process.env.GROQ_API_KEY }); app.post("/api/chat", async (req, res) => { try { const { messages = [], subject = "General", level = "Intermediate", language = "Hinglish" } = req.body; if (!process.env.GROQ_API_KEY) { return res.status(500).json({ error: "Groq API key server par set nahi hai." }); } const safeMessages = Array.isArray(messages) ? messages .filter( msg => msg && ["user", "assistant"].includes(msg.role) && typeof msg.content === "string" ) .slice(-20) : []; const systemPrompt = ` You are School AI, an advanced and friendly school teacher. Subject: ${subject} Student Level: ${level} Language: ${language} Rules: 1. Answer the student's question accurately. 2. Explain difficult topics in simple language. 3. For Mathematics, show formulas and step-by-step calculations. 4. For Physics and Chemistry, explain concepts, formulas and examples. 5. For Biology, explain with clear examples. 6. For programming questions, provide working code and explain it. 7. If the student asks in Hindi, answer in Hindi. 8. If the student asks in Hinglish, answer in Hinglish. 9. Do not unnecessarily make answers complicated. 10. If a question is unclear, ask for clarification. 11. For school homework, teach the concept instead of only giving the final answer. 12. Use headings, bullet points and examples when useful. `; const completion = await groq.chat.completions.create({ model: "llama-3.3-70b-versatile", messages: [ { role: "system", content: systemPrompt }, ...safeMessages ], temperature: 0.3, max_tokens: 2048 }); const reply = completion.choices?.[0]?.message?.content || "Sorry, mujhe answer generate karne mein problem hui."; res.json({ reply }); } catch (error) { console.error("Groq Error:", error); res.status(500).json({ error: "Groq se response nahi mil paya. Server aur API key check karein." }); } }); app.listen(PORT, () => { console.log(`✅ School AI running at http://localhost:${PORT}`); });
